@legal(basis: .public_data, purpose: "price comparison")
fn scrape_prices(urls: []URL) -> Result<[]Price> {
    let pool = IPPool { size: 10_000, rotation: .per_request }

    let pages = pool.map(urls, concurrency: 50) { url ->
        fetch(url, timeout: 5.seconds)?
    }?

    pages.filter  { p -> p.status == 200 }
         .map     { p -> parse_price(p.body)? }
         .flatten
         .encrypt()  // ML-KEM-768 default
}
