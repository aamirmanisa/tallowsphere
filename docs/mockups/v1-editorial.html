Language

A Specimen

@legal(
    basis:         .public_data,
    jurisdiction:  [.us],
    personal_data: false,
    purpose:       "competitive price monitoring",
    retain_for:    7.days,
    robots_txt:    .respect,
    rate_limit:    2.per_second
)
fn search_products(query: String) -> Result<[]Product> {
    let pool = IPPool {
        size:     10_000,
        rotation: .per_request
    }

    let stores = ["heb.com", "kroger.com", "target.com"]
    let urls = stores.map { s -> "https://{s}/search?q={query}" }

    let pages = pool.map(urls, concurrency: 50) { url ->
        fetch(url, timeout: 5.seconds)?
    }?

    pages
        .filter  { p -> p.status == 200 }
        .map     { p -> parse_products(p.body)? }
        .flatten
        .sort_by { p -> p.price }
}
