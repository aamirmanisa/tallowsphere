1. TallowTransfer — File Transfer

Basic File Operations

Hash a File (BLAKE3 — 1,100 MB/s)

REAL CRYPTO1.1 GB/s
tallow hash myfile.dat
# Output: a7f2c3d4e5f6... (64-char hex, BLAKE3)

Fast File Copy (OS-Optimized)

tallow copy source.txt destination.txt
# Preserves modification time, 3-7 GB/s on NVMe

Pack / Unpack Directories (TWD Format)

# Pack a folder into a compressed archive
tallow pack ./my-project project.twd
# Full folder structure preserved, zstd compressed, BLAKE3 integrity

# Unpack
tallow unpack project.twd --output ./restored
# All files, subdirs, nested folders restored exactly

Speed Test

tallow speed-test
# Benchmarks: disk write/read, BLAKE3 hash, AES-256-GCM, zstd compress
#   disk_write_mbps           2847 MB/s

#   blake3_hash_mbps          1122 MB/s

2. File Encryption & Security

Encrypt a Single File

AES-256-GCMARGON2ID
tallow encrypt secrets.db secrets.enc --password "strong-password"
# Argon2id key derivation + AES-256-GCM encryption
# Output: NIXV format (magic header + salt + nonce + ciphertext)

tallow decrypt secrets.enc secrets.db --password "strong-password"

Encrypt Entire Directory

PER-FILE ENCRYPTIONFOLDER STRUCTURE INTACT
tallow encrypt-dir ./private ./encrypted --password "my-password"
# Every file encrypted individually with unique salt+nonce
# Directory structure preserved: ./encrypted/subdir/file.txt.enc

tallow decrypt-dir ./encrypted ./recovered --password "my-password"
# Wrong password? Detected immediately (AES-GCM authentication)

Validate File Type (Magic Bytes)

tallow validate-file suspicious.pdf
# Detects: actual type by magic bytes (not extension!)
# Catches: .exe disguised as .pdf, macros, scripts
# 25+ formats: PNG, JPEG, PDF, ZIP, EXE, ELF, Mach-O, WASM, SQLite...

AV Scanning

tallow scan downloaded-file.exe
# Scans with ClamAV (if installed)
# Also supports YARA rules: tallow scan file --yara rules.yar

3. Delta Sync & Watch Mode

Delta Sync (Only Copy What Changed)

100x FASTERBLAKE3 DIFF
tallow delta-sync ./source ./backup
# Compares by BLAKE3 hash (not timestamp!)
# Only copies files that actually changed
# 100GB folder, 1% changed = copies 1GB, not 100GB

# Preview first (dry run)
tallow delta-sync ./source ./backup --dry-run

Watch Mode (Auto-Sync on Changes)

tallow watch ./my-project --interval 1000
# Polls every 1000ms, detects added/removed/modified files
# Uses BLAKE3 snapshots for change detection

Retention Policy (Auto-Delete Old Files)

tallow retain ./transfers --days 30
# Deletes files older than 30 days

tallow retain ./transfers --days 30 --dry-run
# Preview what would be deleted

4. Share by Link

Share File — Anyone with a Browser Can Download

NO TALLOW NEEDED ON RECEIVER
tallow share report.pdf
# Starts HTTP server, prints download URL:
# http://192.168.1.5:8432/dl/a7f2.../report.pdf
#
# Anyone opens that URL in Chrome/Firefox/Safari/phone → downloads file
# Dark-themed landing page with file info and BLAKE3 hash

# Password protected
tallow share secrets.zip --password "my-secret"

# Limit downloads (self-destructs after 3)
tallow share onetime.pdf --max-downloads 3

# Share entire folder
tallow share ./my-project
# Auto-packs as TWD archive, serves for download

5. Network Transfer (QUIC)

Send with Code Phrase (Like croc)

# Sender: generates code automatically
tallow send file.zip
# Code: 742-falcon-ember-crystal
# QR code displayed for mobile scanning
# Share the code with your recipient

# Receiver: type the code
tallow receive 742-falcon-ember-crystal

Direct Transfer (Know the IP)

# Machine A: listen for incoming
tallow receive --listen 0.0.0.0:6881

# Machine B: send directly
tallow send file.zip --to 192.168.1.5:6881
# QUIC protocol (TLS 1.3 encrypted, BLAKE3 verified)

Multi-File + Folder Send

tallow send file1.txt file2.pdf ./mydir --to 192.168.1.5:6881 --compress
# Multiple files + directories in one transfer
# Auto-compressed with zstd, folder structure preserved (TWD format)

Send Text (No File Needed)

tallow send-text "Hello from my PC!"
# Sends text directly, BLAKE3 hashed

Relay Server (When Direct Connection Fails)

# Start your own relay
tallow relay --port 6880

# Send via relay
tallow send file.zip --relay relay.example.com:6880

LAN Discovery

tallow discover
# Finds devices on your local network
# Shows: IP, hostname, port for direct transfer

6. Encrypted Chat & Group Chat

1-to-1 Encrypted Chat

AES-256-GCMPER-MESSAGE ENCRYPTION
# Alice starts a chat
tallow chat --name "Alice"
# Code: 531-crystal-phoenix-dawn

# Bob joins
tallow chat --name "Bob" --code 531-crystal-phoenix-dawn
# Each message encrypted with unique AES-256-GCM nonce

Group Chat (Up to 50 Members)

tallow group-chat --name "Team Alpha" --user "Alice"
# Per-group encryption key derived from group ID
# Members: add/remove with GroupChat.add_member/remove_member

9. Web Scraping

Complete Scraper Example

// scraper.tl
@legal(basis: .public_data, purpose: "price monitoring",
       retain_for: 7.days, robots_txt: .respect)
fn main() {
    let browser = Browser.launch(true)
    let page = browser.new_page()
    page.apply_stealth()
    page.block_resources()

    page.goto("https://books.toscrape.com")

    // ONE CDP call extracts ALL prices (not N calls)
    let prices = page.extract_all_text(".price_color")
    let titles = page.extract_all_text("h3 a")

    for i in 0..len(titles) {
        print("{titles[i]} — {prices[i]}")
    }

    browser.close()
}
tallow run scraper.tl
